5. DDL 스크립트

Hibernate ddl-auto: update가 자동 관리하지만, 초기 구성 참조용입니다.

-- users
CREATE TABLE users (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email           VARCHAR(255) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,
    name            VARCHAR(255) NOT NULL,
    phone_number    VARCHAR(255),
    role            VARCHAR(255) DEFAULT 'USER',
    mobile_carrier  VARCHAR(255) NOT NULL,  -- SKT / SKT 알뜰폰 / KT / KT 알뜰폰 / LG U+ / LG U+ 알뜰폰
    created_at      TIMESTAMP NOT NULL DEFAULT now(),
    updated_at      TIMESTAMP
);

-- subscription_categories
CREATE TABLE subscription_categories (
    id         UUID PRIMARY KEY,
    name       VARCHAR(50) NOT NULL UNIQUE,
    icon       VARCHAR(100),
    color      VARCHAR(7),
    is_default BOOLEAN NOT NULL DEFAULT FALSE
);

-- subscriptions
CREATE TABLE subscriptions (
    id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    category_id           UUID REFERENCES subscription_categories(id) ON DELETE SET NULL,
    service_id            UUID REFERENCES subscription_services(id),
    bundle_id             UUID REFERENCES bundle_catalog(id) ON DELETE SET NULL,
    service_name          VARCHAR(100) NOT NULL,
    description           TEXT,
    amount                NUMERIC(12,2) NOT NULL,
    currency              VARCHAR(10) NOT NULL DEFAULT 'KRW',
    billing_cycle         VARCHAR(10) NOT NULL DEFAULT 'MONTHLY',
    start_date            DATE DEFAULT now(),
    next_billing_date     DATE NOT NULL,
    billing_day_of_month  INTEGER,
    billing_day_of_week   INTEGER,
    billing_date_of_year  VARCHAR(4),
    status                VARCHAR(10) NOT NULL DEFAULT 'ACTIVE',
    is_bundle             BOOLEAN NOT NULL DEFAULT FALSE,
    notify_before         BOOLEAN NOT NULL DEFAULT TRUE,
    notify_days_before    SMALLINT NOT NULL DEFAULT 3,
    created_at            TIMESTAMP NOT NULL DEFAULT now(),
    updated_at            TIMESTAMP
);

-- service_categories
CREATE TABLE service_categories (
    id            UUID PRIMARY KEY,
    name          VARCHAR(50) NOT NULL UNIQUE,
    display_order INTEGER NOT NULL
);

-- subscription_services
CREATE TABLE subscription_services (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name          VARCHAR(100) NOT NULL,
    category_id   UUID NOT NULL REFERENCES service_categories(id),
    logo_domain   VARCHAR(100) NOT NULL,
    display_order INTEGER NOT NULL,
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_at    TIMESTAMP NOT NULL DEFAULT now()
);

-- bundle_catalog
CREATE TABLE bundle_catalog (
    id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    provider         VARCHAR(100) NOT NULL,
    provider_url     TEXT,
    plan_name        VARCHAR(100) NOT NULL,
    plan_url         TEXT,
    description      TEXT,
    includes         TEXT[] NOT NULL,
    service_ids      UUID[],
    base_price       INTEGER NOT NULL,
    original_price   INTEGER,
    discount_rate    INTEGER,
    contract_months  INTEGER,
    telecom_exclusive VARCHAR(50),
    has_options      BOOLEAN NOT NULL DEFAULT FALSE,
    is_active        BOOLEAN NOT NULL DEFAULT TRUE,
    crawled_at       TIMESTAMP NOT NULL DEFAULT now(),
    created_at       TIMESTAMP NOT NULL DEFAULT now(),
    updated_at       TIMESTAMP
);
CREATE INDEX idx_bundle_catalog_provider   ON bundle_catalog USING btree (provider);
CREATE INDEX idx_bundle_catalog_is_active  ON bundle_catalog USING btree (is_active);
CREATE INDEX idx_bundle_catalog_includes   ON bundle_catalog USING gin  (includes);

-- bundle_option
CREATE TABLE bundle_option (
    id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bundle_id    UUID NOT NULL REFERENCES bundle_catalog(id) ON DELETE CASCADE,
    option_name  VARCHAR(200) NOT NULL,
    option_spec  VARCHAR(300),
    extra_price  INTEGER NOT NULL DEFAULT 0,
    total_price  INTEGER NOT NULL,
    is_active    BOOLEAN NOT NULL DEFAULT TRUE,
    created_at   TIMESTAMP NOT NULL DEFAULT now(),
    UNIQUE (bundle_id, option_name)
);

-- bundle_price_history
CREATE TABLE bundle_price_history (
    id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bundle_id      UUID NOT NULL REFERENCES bundle_catalog(id) ON DELETE CASCADE,
    old_price      INTEGER NOT NULL,
    new_price      INTEGER NOT NULL,
    change_reason  VARCHAR(100),
    changed_at     TIMESTAMP NOT NULL DEFAULT now()
);

-- document_chunk (RAG용 벡터 임베딩 — pgvector 확장 필요)
CREATE TABLE document_chunk (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    source     VARCHAR(100) NOT NULL,
    source_id  UUID,
    content    TEXT NOT NULL,
    embedding  vector,
    created_at TIMESTAMP NOT NULL DEFAULT now()
);
CREATE INDEX idx_document_chunk_embedding ON document_chunk USING ivfflat (embedding vector_cosine_ops) WITH (lists='100');

-- ai_sessions
CREATE TABLE ai_sessions (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id    UUID NOT NULL REFERENCES users(id),
    title      VARCHAR(100) NOT NULL,
    messages   JSONB NOT NULL DEFAULT '[]',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

6. 운영 참고사항

6.1 ddl-auto: update 주의

  • 컬럼 추가는 자동으로 적용됩니다.
  • 컬럼 삭제/타입 변경은 자동으로 반영되지 않으며, 수동 ALTER TABLE이 필요합니다.
  • 예: currency 컬럼의 커스텀 타입 currency_code를 VARCHAR로 변경 시 수동 마이그레이션 필요했음.

6.2 Redis 데이터 (비관계형)
